1  class wr_busfm;
     2    virtual intf vif;
     3    wr_transaction wr_tx;
     4  
     5    function new();
     6      vif = top.pif;
     7    endfunction
     8  
     9   task run();
     10    forever begin
     11      @(posedge vif.wr_clk_i);
     12      if (vif.rst_i == 1) begin   // sample rst_i synchronously with clock
     13        gen2bfm_wr.get(wr_tx);
     14        fork
     15          begin : DRIVE
     16            drive_tx(wr_tx);   // edge-aligned version, no leading @(posedge)
     17          end
     18          begin : RESET_WATCH
     19            wait(vif.rst_i == 0);
     20            vif.wr_en_i = 0;
     21          end
     22        join_any
     23        disable fork;
     24      end
     25    end
     26  endtask  
     27  
     28    task drive_tx(wr_transaction wr_tx);
     29      @(posedge vif.wr_clk_i);
     30      vif.wr_en_i = wr_tx.wr_en_i;
     31      if (wr_tx.wr_en_i == 1)
     32        vif.wdata_i = wr_tx.wdata_i;
     33      @(posedge vif.wr_clk_i);
     34      common::wr_bfm_count++;
     35      vif.wr_en_i = 0;
     36    endtask
     37  endclass
