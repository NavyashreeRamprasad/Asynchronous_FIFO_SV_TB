1  module async_fifo (
     2      input  wr_clk_i,
     3      input  rd_clk_i,
     4      input  rst_i,
     5      input  wr_en_i,
     6      input  rd_en_i,
     7      input  [`DATA_WIDTH-1:0] wdata_i,
     8      output reg [`DATA_WIDTH-1:0] rdata_o,
     9      output reg full_o,
     10      output reg empty_o,
     11      output reg overflow_o,
     12      output reg underflow_o
     13  );
     14  
     15  integer i;
     16  
     17  // FIFO Memory
     18  reg [`DATA_WIDTH-1:0] fifo [0:`DEPTH-1];
     19  
     20  // Binary pointers
     21  reg [`PTR_WIDTH-1:0] wr_pntr, rd_pntr;
     22  
     23  // Toggle bits
     24  reg wr_toggle, rd_toggle;
     25  
     26  // Pointer synchronization
     27  reg [`PTR_WIDTH-1:0] rd_pntr_wr_clk;
     28  reg [`PTR_WIDTH-1:0] wr_pntr_rd_clk;
     29  
     30  reg rd_toggle_wr_clk;
     31  reg wr_toggle_rd_clk;
     32  
     33  // Write Logic
     34  
     35  always @(posedge wr_clk_i or negedge rst_i)
     36  begin
     37      if(rst_i==0) begin
     38          wr_pntr      <= 0;
     39          wr_toggle    <= 0;
     40          overflow_o   <= 0;
     41  
     42          for(i=0;i<`DEPTH;i=i+1)
     43              fifo[i] <= 0;
     44      end
     45      else begin
     46  
     47          overflow_o <= 0;
     48  
     49          if(wr_en_i) begin
     50  
     51              if(full_o)
     52                  overflow_o <= 1;
     53  
     54              else begin
     55  
     56                  fifo[wr_pntr] <= wdata_i;
     57  
     58                  if(wr_pntr == `DEPTH-1) begin
     59                      wr_pntr   <= 0;
     60                      wr_toggle <= ~wr_toggle;
     61                  end
     62                  else
     63                      wr_pntr <= wr_pntr + 1;
     64  
     65              end
     66          end
     67      end
     68  end
     69  
     70  // Read Logic
     71  
     72  always @(posedge rd_clk_i or negedge rst_i)
     73  begin
     74      if(rst_i==0) begin
     75          rd_pntr      <= 0;
     76          rd_toggle    <= 0;
     77          rdata_o      <= 0;
     78          underflow_o  <= 0;
     79      end
     80      else begin
     81  
     82          underflow_o <= 0;
     83  
     84          if(rd_en_i) begin
     85  
     86              if(empty_o)
     87                  underflow_o <= 1;
     88  
     89              else begin
     90  
     91                  rdata_o <= fifo[rd_pntr];
     92  
     93                  if(rd_pntr == `DEPTH-1) begin
     94                      rd_pntr   <= 0;
     95                      rd_toggle <= ~rd_toggle;
     96                  end
     97                  else
     98                      rd_pntr <= rd_pntr + 1;
     99  
     100              end
     101          end
     102      end
     103  end
     104  
     105  // Synchronize Read Pointer into Write Clock
     106  
     107  always @(posedge wr_clk_i or negedge rst_i)
     108  begin
     109      if(rst_i==0) begin
     110          rd_pntr_wr_clk   <= 0;
     111          rd_toggle_wr_clk <= 0;
     112      end
     113      else begin
     114          rd_pntr_wr_clk   <= rd_pntr;
     115          rd_toggle_wr_clk <= rd_toggle;
     116      end
     117  end
     118  
     119  // Synchronize Write Pointer into Read Clock
     120  
     121  always @(posedge rd_clk_i or negedge rst_i)
     122  begin
     123      if(rst_i==0) begin
     124          wr_pntr_rd_clk   <= 0;
     125          wr_toggle_rd_clk <= 0;
     126      end
     127      else begin
     128          wr_pntr_rd_clk   <= wr_pntr;
     129          wr_toggle_rd_clk <= wr_toggle;
     130      end
     131  end
     132  
     133  // Full Flag
     134  
     135  always @(*)
     136  begin
     137      if(rst_i==0)
     138          full_o <= 0;
     139      else
     140          full_o <= (wr_pntr == rd_pntr_wr_clk) &&
     141                    (wr_toggle != rd_toggle_wr_clk);
     142  end
     143  
     144  // Empty Flag
     145  
     146  always @(*)
     147  begin
     148      if(rst_i==0)
     149          empty_o <= 1;
     150      else
     151          empty_o <= (wr_pntr_rd_clk == rd_pntr) &&
     152                     (wr_toggle_rd_clk == rd_toggle);
     153  end
     154  
     155  endmodule
